e=>`/* Scaling a translucent tile over translucent glass looks
   cheap, so the dock scales the glyph and has no tile behind
   it at all. --d is the item's distance from the cursor, set
   on each item in JS; everything else is here. */
.gdock-glyph {
  /* a raised cosine over the reach: 1 under the cursor, 0 at
     the edge of it, and nothing at all past it */
  --f: max(0, (1 + cos(180deg * var(--d) / ${Number(e.spread)+1})) / 2);
  transform: translateY(calc(${xy(Number(e.lift))}px * var(--f) * -1))
             scale(calc(1 + ${xy(Number(e.magnify))} * var(--f) - var(--f)));
  transform-origin: 50% 100%;
  transition: transform 220ms cubic-bezier(0.32, 0.72, 0.24, 1);
}

/* the glyph magnifies, not the button. Scaling the whole item
   drags the running dot and the label along with it, and the
   label ends up 15px further from the bar purely as a side
   effect of the zoom. */
.gdock-item { transform: none; }

.gdock-tip { display: ${e.labels?`block`:`none`}; }`
